#include <stdio.h>

#include "openvino/genai/c/text2speech_pipeline.h"

int main(int argc, char** argv) {
    if (argc != 2) return 1;

    ov_genai_text2speech_pipeline* pipeline = NULL;
    ov_genai_text2speech_decoded_results* results = NULL;
    ov_tensor_t* speech = NULL;
    FILE* output = NULL;
    int status = 1;

    if (ov_genai_text2speech_pipeline_create(argv[1], "${props.device || 'CPU'}", 0, &pipeline) != OK ||
        ov_genai_text2speech_pipeline_generate(pipeline, "Hello OpenVINO GenAI", NULL, NULL, &results) != OK ||
        ov_genai_text2speech_decoded_results_get_speech_at(results, 0, &speech) != OK)
        goto cleanup;

    size_t sample_count = 0;
    uint32_t sample_rate = 0;
    void* samples = NULL;
    if (ov_tensor_get_size(speech, &sample_count) != OK ||
        ov_tensor_data(speech, &samples) != OK ||
        ov_genai_text2speech_decoded_results_get_output_sample_rate(results, &sample_rate) != OK)
        goto cleanup;

    output = fopen("speech.f32", "wb");
    if (!output || fwrite(samples, sizeof(float), sample_count, output) != sample_count)
        goto cleanup;
    printf("Saved %zu float32 samples at %u Hz\\n", sample_count, sample_rate);
    status = 0;

cleanup:
    if (output) fclose(output);
    ov_tensor_free(speech);
    ov_genai_text2speech_decoded_results_free(results);
    ov_genai_text2speech_pipeline_free(pipeline);
    return status;
}
