import { readImages, readVideo } from "./utils.js";

const imageEmbedding = await pipeline.embed("", { images });

// Embed a video (video is an ov.Tensor of shape [F, H, W, 3], uint8)
const [video, videoMetadata] = await readVideo("path/to/video.mp4");
const videoEmbedding = await pipeline.embed("", { videos: [video], videosMetadata: [videoMetadata] });
`}
