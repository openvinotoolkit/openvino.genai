# Streaming the Output

For more interactive UIs during generation, you can stream output tokens.

:::info
Streaming is supported for `LLMPipeline`, `VLMPipeline` and `ASRPipeline`.
:::

## Streaming Function

In this example, a function outputs words to the console immediately upon generation:

<LanguageTabs>
    <TabItemPython>
        ```python showLineNumbers
        import openvino_genai as ov_genai

        pipe = ov_genai.LLMPipeline(model_path, "CPU")

        # highlight-start
        # Create a streamer function
        def streamer(subword):
            print(subword, end='', flush=True)
            # Return flag corresponds whether generation should be stopped.
            return ov_genai.StreamingStatus.RUNNING
        # highlight-end

        # highlight-next-line
        pipe.start_chat()
        while True:
            try:
                prompt = input('question:\n')
            except EOFError:
                break
            # highlight-next-line
            pipe.generate(prompt, streamer=streamer, max_new_tokens=100)
            print('\n----------\n')
        # highlight-next-line
        pipe.finish_chat()
        ```
    </TabItemPython>
    <TabItemCpp>
        ```cpp showLineNumbers
        #include "openvino/genai/llm_pipeline.hpp"
        #include <iostream>

        int main(int argc, char* argv[]) {
            std::string prompt;

            std::string model_path = argv[1];
            ov::genai::LLMPipeline pipe(model_path, "CPU");

            // highlight-start
            // Create a streamer function
            auto streamer = [](std::string word) {
                std::cout << word << std::flush;
                // Return flag corresponds whether generation should be stopped.
                return ov::genai::StreamingStatus::RUNNING;
            };
            // highlight-end

            // highlight-next-line
            pipe.start_chat();
            std::cout << "question:\n";
            while (std::getline(std::cin, prompt)) {
                // highlight-next-line
                pipe.generate(prompt, ov::genai::streamer(streamer), ov::genai::max_new_tokens(100));
                std::cout << "\n----------\n"
                    "question:\n";
            }
            // highlight-next-line
            pipe.finish_chat();
        }
        ```
    </TabItemCpp>
    <TabItemJS>
        ```js showLineNumbers
        import { LLMPipeline } from "openvino-genai-node";
        import readline from 'readline';

        const pipe = await LLMPipeline(model_path, "CPU");

        // highlight-start
        // Create a streamer function
        function streamer(subword) {
            process.stdout.write(subword);
        }
        // highlight-end

        // highlight-next-line
        await pipe.startChat();

        const rl = readline.createInterface({
            input: process.stdin,
            output: process.stdout,
        });

        console.log('question:');
        rl.on('line', async (prompt) => {
            // highlight-next-line
            await pipe.generate(prompt, { max_new_tokens: 100 }, streamer);
            console.log('\n----------\nquestion:');
        });

        rl.on('close', async () => {
            // highlight-next-line
            await pipe.finishChat();
            process.exit(0);
        });
        ```

## Custom Streamer Class

You can also create your custom streamer for more sophisticated processing:

<LanguageTabs>
    <TabItemPython>
        ```python showLineNumbers
        import openvino_genai as ov_genai

        pipe = ov_genai.LLMPipeline(model_path, "CPU")

        # highlight-start
        # Create custom streamer class
        class CustomStreamer(ov_genai.StreamerBase):
            def __init__(self):
                super().__init__()
                # Initialization logic.

            def write(self, token: int | list[int]) -> ov_genai.StreamingStatus:
                # Custom processing logic for new decoded token(s).

                # Return flag corresponds whether generation should be stopped.
                return ov_genai.StreamingStatus.RUNNING

            def end(self):
                # Custom finalization logic.
                pass
        # highlight-end

        # highlight-next-line
        pipe.start_chat()
        while True:
            try:
                prompt = input('question:\n')
            except EOFError:
                break
            # highlight-next-line
            pipe.generate(prompt, streamer=CustomStreamer(), max_new_tokens=100)
            print('\n----------\n')
        # highlight-next-line
        pipe.finish_chat()
        ```
    </TabItemPython>
    <TabItemCpp>
        ```cpp showLineNumbers
        #include "openvino/genai/streamer_base.hpp"
        #include "openvino/genai/llm_pipeline.hpp"
        #include <iostream>

        // highlight-start
        // Create custom streamer class
        class CustomStreamer: public ov::genai::StreamerBase {
        public:
            ov::genai::StreamingStatus write(int64_t token) {
                // Custom processing logic for new decoded token.

                // Return flag corresponds whether generation should be stopped.
                return ov::genai::StreamingStatus::RUNNING;
            };

            ov::genai::StreamingStatus write(const std::vector<int64_t>& tokens) {
                // Custom processing logic for new vector of decoded tokens.

                // Return flag corresponds whether generation should be stopped.
                return ov::genai::StreamingStatus::RUNNING;
            };

            void end() {
                // Custom finalization logic.
            };
        };
        // highlight-end

        int main(int argc, char* argv[]) {
            std::string prompt;
            // highlight-next-line
            std::shared_ptr<CustomStreamer> custom_streamer;

            std::string model_path = argv[1];
            ov::genai::LLMPipeline pipe(model_path, "CPU");

            // highlight-next-line
            pipe.start_chat();
            std::cout << "question:\n";
            while (std::getline(std::cin, prompt)) {
                // highlight-next-line
                pipe.generate(prompt, ov::genai::streamer(custom_streamer), ov::genai::max_new_tokens(100));
                std::cout << "\n----------\n"
                    "question:\n";
            }
            // highlight-next-line
            pipe.finish_chat();
        }
        ```

## Monitoring token generation in streaming ASR

`ASRStream.get_num_generated_tokens()` returns the number of generated tokens so far and is safe to call from the
transcript callback. Sampling the count with a monotonic timer lets applications display throughput while audio is
being transcribed:

For Voxtral, `ASRStream.get_kv_cache_size()` reports the combined retained audio and text KV-cache tensor bytes
after the latest completed inference chunk, including sliding-window cropping. It starts at zero and is safe to
query during generation. The count excludes convolution caches, inference request buffers, model weights and
temporary allocations; it is not process memory or peak inference memory. Unsupported ASR models raise an exception.

`ASRStream.get_kv_cache_window_sizes()` returns `audio` and `text` entries, each containing
`(retained_positions, maximum_positions)` in Python or a `std::pair` in C++. Retained sizes start at zero and
are sampled from KV tensor sequence lengths after the latest completed chunk and cropping. Maxima come from
the loaded model configuration. The snapshot is safe to query during generation, without enabling operation
profiling; unsupported models raise an exception. These positions are not transcript word counts.
The Voxtral sample displays both windows as `retained/max (percent%)` alongside KV memory usage.

For Voxtral, `ASRStream.get_stage_durations()` returns a thread-safe snapshot of cumulative milliseconds with keys
`audio_inference`, `text_inference`, `audio_kv_cache` and `text_kv_cache`. Inference timings cover `infer()` calls.
KV timings cover retained tensor allocation, copying, cropping and rephasing. Feature extraction, convolution cache
copies, tensor setup, sampling and callbacks are excluded. Values start at zero and remain available after generation.
For interval milliseconds per generated step, subtract the previous snapshot and divide by the number of new
generated tokens. The first interval includes prefill. These stages do not account for the entire wall-clock time.
Unsupported models raise an exception.

When constructing a Voxtral pipeline with `PERF_COUNT=True` (Python) or `ov::enable_profiling(true)` (C++),
`get_stage_durations()` also includes `audio_op/<node_type>/<exec_type>` and `text_op/<node_type>/<exec_type>`
keys. Each groups the latest plugin real-time counters in milliseconds. CPU counters are running averages across
node executions, not per-call or cumulative durations; do not accumulate, difference or divide them by token counts.
`audio_node/<description>` and `text_node/<description>` expose individual executed Broadcast counters with full
node names, execution types, compiled-graph input/output partial shapes (`?` denotes dynamic) and consumer names. These are graph
shapes, not runtime tensor measurements. Unmatched plugin node names are marked explicitly. The sample shows the
three most expensive Broadcast nodes per component. Fused operations remain fused;
counter sums may overlap or omit overhead and need not equal inference wall time. Profiling adds overhead and
is disabled by default. The Voxtral streaming sample's `--profile` flag displays the top eight groups per component.

<TabItemPython>

```python
import time
import openvino_genai as ov_genai

def make_callback(stream, started_at):
    last_time = started_at
    last_tokens = 0
    last_stages = stream.get_stage_durations()

    def on_text(delta):
        nonlocal last_time, last_tokens, last_stages
        now = time.perf_counter()
        tokens = stream.get_num_generated_tokens()
        interval = now - last_time
        elapsed = now - started_at
        interval_rate = (tokens - last_tokens) / interval if interval > 0 else 0.0
        average_rate = tokens / elapsed if elapsed > 0 else 0.0
        print(delta, end="", flush=True)
        cache_mib = stream.get_kv_cache_size() / (1024 ** 2)
        for component, (used, maximum) in stream.get_kv_cache_window_sizes().items():
            print(f"{component} window: {used}/{maximum} ({100 * used / maximum:.1f}%)")
        print(
            f"\n{tokens} tokens | {interval_rate:.1f} tok/s interval | "
            f"{average_rate:.1f} tok/s avg | KV cache: {cache_mib:.1f} MiB",
            flush=True,
        )
        stages = stream.get_stage_durations()
        new_tokens = tokens - last_tokens
        if new_tokens > 0:
            print("interval ms/step: " + " | ".join(
                f"{stage}={(duration - last_stages[stage]) / new_tokens:.2f}"
                for stage, duration in stages.items() if stage in {
                    "audio_inference", "text_inference", "audio_kv_cache", "text_kv_cache"
                }
            ), flush=True)
        last_stages = stages
        last_time, last_tokens = now, tokens
        return ov_genai.StreamingStatus.RUNNING

    return on_text

started_at = time.perf_counter()
text_callback = make_callback(stream, started_at)
stream.generate(text_callback)
```

const auto started_at = std::chrono::steady_clock::now();
auto last_time = started_at;
size_t last_tokens = 0;
auto last_stages = stream.get_stage_durations();
const auto result = stream.generate([&](std::string delta) {
    const auto now = std::chrono::steady_clock::now();
    const size_t tokens = stream.get_num_generated_tokens();
    const double interval = std::chrono::duration<double>(now - last_time).count();
    const double elapsed = std::chrono::duration<double>(now - started_at).count();
    const double interval_rate = interval > 0.0 ? (tokens - last_tokens) / interval : 0.0;
    const double average_rate = elapsed > 0.0 ? tokens / elapsed : 0.0;
    const double cache_mib = stream.get_kv_cache_size() / (1024.0 * 1024.0);
    for (const auto& [component, window] : stream.get_kv_cache_window_sizes())
        std::cout << component << " window: " << window.first << "/" << window.second
                  << " (" << 100.0 * window.first / window.second << "%)" << std::endl;
    std::cout << delta << "\n" << tokens << " tokens | " << interval_rate
              << " tok/s interval | " << average_rate << " tok/s avg | KV cache: "
              << cache_mib << " MiB" << std::endl;
    const auto stages = stream.get_stage_durations();
    if (tokens > last_tokens) {
        std::cout << "interval ms/step:";
        for (const std::string stage : {"audio_inference", "text_inference", "audio_kv_cache", "text_kv_cache"})
            std::cout << " | " << stage << "=" << (stages.at(stage) - last_stages.at(stage)) / (tokens - last_tokens);
        std::cout << std::endl;
    }
    last_stages = stages;
    last_time = now;
    last_tokens = tokens;
    return ov::genai::StreamingStatus::RUNNING;
});
std::cout << result.texts.front() << std::endl;
